cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
project(OpenWillow LANGUAGES C CXX)
option(OPENWILLOW_RESEARCH_LZO "Enable GPL-2.0-or-later miniLZO in the local research reader" OFF)
add_executable(ow-package src/package.cpp src/container.cpp)
target_include_directories(ow-package PRIVATE src)
if(OPENWILLOW_RESEARCH_LZO)
  include(FetchContent)
  FetchContent_Declare(minilzo
    URL https://www.oberhumer.com/opensource/lzo/download/minilzo-2.10.tar.gz
    URL_HASH SHA256=eb4ce543aad19533c83550746e0e9d7bcf716b35a42429e3ba17d60fa0f3e47a)
  FetchContent_MakeAvailable(minilzo)
  add_library(ow-minilzo STATIC ${minilzo_SOURCE_DIR}/minilzo.c)
  target_include_directories(ow-minilzo PUBLIC ${minilzo_SOURCE_DIR})
  target_link_libraries(ow-package PRIVATE ow-minilzo)
  target_compile_definitions(ow-package PRIVATE OW_RESEARCH_LZO=1)
endif()
target_compile_features(ow-package PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(ow-package PRIVATE /W4 /permissive-)
else()
  target_compile_options(ow-package PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME package-synthetic COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/package_test.py $<TARGET_FILE:ow-package>)
if(OPENWILLOW_RESEARCH_LZO)
  add_test(NAME container-synthetic COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/package_test.py $<TARGET_FILE:ow-package> --compressed)
endif()
